# Тесты для варианта 19: вынести из частных унарные минусы

# Базовые случаи
(-a) / b
a / (-b)
(-a) / (-b)

# С константами
(-5) / 3
4 / (-2)
(-3) / (-7)

# Без изменений (нет унарных минусов в делении)
a / b
a + b
a * b

# Сложные случаи
a * b / (-c)
(-a) / b + c
a / (-b) * c
(-a) / (-b) + d
(-(a + b)) / c
a / ((-b) * c)
(-a) * b / ((-c) * d)
